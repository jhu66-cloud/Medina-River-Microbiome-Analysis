------------------PCoA plot to visualize seasonal and spatial difference of river water samples based on bray-curtis distance-------------------------
-----------------PCoA plot with hull interior colors to distinguish groups-----------------------------------------------

library(phyloseq)
library(ggplot2)
library(dplyr)
library(vegan)

## ---------------------------
## 1. Read distance & metadata
## ---------------------------

bray_curtis_dis <- read.csv(
  "bray-curtis-distance-water.csv",
  sep = ",",
  header = TRUE,
  row.names = 1
)

sampledata <- read.csv(
  "Metadata_Water.csv",
  sep = ",",
  header = TRUE,
  row.names = 1
)

bray_curtis <- as.dist(bray_curtis_dis)

## ---------------------------
## 2. PCoA
## ---------------------------

ordination <- cmdscale(bray_curtis, eig = TRUE, k = 3)

eigenvalues <- ordination$eig
variance_explained <- eigenvalues / sum(eigenvalues)

PC1_variance <- round(variance_explained[1] * 100, 2)
PC2_variance <- round(variance_explained[2] * 100, 2)
PC3_variance <- round(variance_explained[3] * 100, 2)

pcoa_scores <- as.data.frame(ordination$points)
colnames(pcoa_scores) <- c("PC1", "PC2", "PC3")

pcoa_scores$SampleID <- rownames(pcoa_scores)

pcoa_scores <- merge(
  pcoa_scores,
  sampledata,
  by = "row.names"
)

## ---------------------------
## 3. Force correct Site order
## ---------------------------

site_order <- c(
  "MP301", "MR306", "MR304", "MR207", "MR205",
  "MR203", "MR102", "MR1015", "MR101"
)

pcoa_scores$Site <- factor(pcoa_scores$Site, levels = site_order)

## ==========PC1~PC2==============
## ---------------------------
## 4. Compute convex hulls
## ---------------------------

hull_data <- pcoa_scores %>%
  group_by(Site) %>%
  slice(chull(PC1, PC2)) %>%
  ungroup()

## ---------------------------
## 5. Define Site colors
## ---------------------------

site_colors <- c(
  "#2166AC","#4393C3","#92C5DE","#C6DBEF","#D9D9D9",
  "#FDDBC7","#F4A582","#D6604D","#B2182B"
)

## ---------------------------
## 6. Plot
## ---------------------------

ggplot(pcoa_scores, aes(x = PC1, y = PC2, color = Site)) +

  ## Hulls (transparent interior, same color as points)
  geom_polygon(
    data = hull_data,
    aes(fill = Site, group = Site),
    alpha = 0.20,
    color = NA
  ) +

  ## Points
  geom_point(size = 3.5) +

  ## Ensure identical mapping for color and fill
  scale_color_manual(
    values = site_colors,
    limits = site_order
  ) +
  scale_fill_manual(
    values = site_colors,
    limits = site_order,
    guide = "none"
  ) +

  ## Axis labels
  xlab(paste("PC1 (", PC1_variance, "%)", sep = "")) +
  ylab(paste("PC2 (", PC2_variance, "%)", sep = "")) +

  ## Theme
  theme(
    text = element_text(family = "Arial"),
    panel.border = element_rect(colour = "black", fill = NA, size = 0.5),
    axis.text = element_text(size = 13),
    axis.title = element_text(size = 15),
    legend.title = element_text(size = 13),
    legend.text = element_text(size = 12),
    legend.background = element_blank(),
    legend.key = element_blank(),
    plot.margin = unit(c(1, 1, 1, 1), "cm"),
    panel.grid = element_line(color = "white", size = 0.1),
    panel.background = element_rect(fill = "white")
  ) +

  ## Zero reference lines
  geom_vline(
    xintercept = 0,
    size = 0.2,
    color = "grey60",
    linetype = "dashed"
  ) +
  geom_hline(
    yintercept = 0,
    size = 0.2,
    color = "grey60",
    linetype = "dashed"
  )
ggsave("PCoA-water-site-PC12-bray-curtis.tiff", width = 5000, height = 3805, units = "px", dpi = 600)
