16S rRNA Amplicon Sequencing Data Processing
===========================================

Software: QIIME 2 (version 2024.10-Amplicon)
Denoising: DADA2
Input: Paired-end Illumina FASTQ files

# 1. Import paired-end sequencing data

qiime tools import \
  --type 'SampleData[PairedEndSequencesWithQuality]' \
  --input-path manifest.txt \
  --output-path paired-end-demux.qza \
  --input-format PairedEndFastqManifestPhred33


# 2. Summarize sequence quality

qiime demux summarize \
  --i-data paired-end-demux.qza \
  --o-visualization paired-end-demux.qzv


# 3. DADA2 denoising, quality filtering,
#    paired-end merging, and chimera removal

qiime dada2 denoise-paired \
  --p-n-threads 16 \
  --i-demultiplexed-seqs paired-end-demux.qza \
  --p-trim-left-f 19 \
  --p-trim-left-r 23 \
  --p-trunc-len-f 280 \
  --p-trunc-len-r 260 \
  --o-table feature_table.qza \
  --o-representative-sequences rep-seqs.qza \
  --o-denoising-stats denoising_stats.qza


# 4. Remove ASVs with fewer than 2 total reads

qiime feature-table filter-features \
  --i-table feature_table.qza \
  --p-min-frequency 2 \
  --o-filtered-table filtered_feature_table.qza

qiime feature-table filter-seqs \
  --i-data rep-seqs.qza \
  --i-table filtered_feature_table.qza \
  --o-filtered-data filtered_rep-seqs.qza


# 5. Summarize filtered ASV table and denoising statistics

qiime feature-table summarize \
  --i-table filtered_feature_table.qza \
  --o-visualization filtered_feature_table.qzv

qiime metadata tabulate \
  --m-input-file denoising_stats.qza \
  --o-visualization denoising_stats.qzv


# 6. Export ASV abundance table

qiime tools export \
  --input-path filtered_feature_table.qza \
  --output-path exported_table

biom convert \
  -i exported_table/feature-table.biom \
  -o ASV_table.tsv \
  --to-tsv


# 7. Export representative ASV sequences

qiime tools export \
  --input-path filtered_rep-seqs.qza \
  --output-path exported_sequences

# Exported FASTA: exported_sequences/dna-sequences.fasta


# Notes
# - Manifest file contains sample IDs and paired-end FASTQ paths.
# - DADA2 parameters reflect the settings used in this study.
# - ASVs with total frequency <2 were removed.
# - Output: ASV abundance table and representative sequences.
